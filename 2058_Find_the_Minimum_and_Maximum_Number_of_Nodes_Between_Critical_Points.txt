                 ans.append(i)
        min = -1
        max = -1
        if len(ans) > 1 :
             max = ans[len(ans)-1] - ans[0]
             min = ans[1]-ans[0]
             for j in range(2,len(ans)):
                 if ans[j]-ans[j-1]<min:
                     min = ans[j]-ans[j-1]

        return [min,max]
