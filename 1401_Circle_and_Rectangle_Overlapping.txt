        closestX = max(x1, min(xCenter, x2))

        # Closest y-coordinate inside rectangle
        closestY = max(y1, min(yCenter, y2))

        # Distance from circle center to closest point
        dx = xCenter - closestX
        dy = yCenter - closestY

        # Check if distance <= radius
        return dx * dx + dy * dy <= radius * radius
